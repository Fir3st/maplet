Your first HTML plugin

const plugin = map.use(({
  container, onCleanup
}) => {
  const el = document
    .createElement('span');
  el.textContent = 'My plugin';
  container.append(el);
  onCleanup(() => el.remove());
  return { element: el };
});

plugin.api.element; // typed API
plugin.remove();    // cleanup
